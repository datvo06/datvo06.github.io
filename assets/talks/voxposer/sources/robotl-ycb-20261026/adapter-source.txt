interfaces.py), and RoboTL ff60eaad from
../RoboTL/.claude/worktrees/leap-codesign-v2/scripts. LMP.exec_safe re-raises
execution exceptions, so TrialEnded unwinds to this runner's finalization.
"""

class TrialEnded(Exception):
    """The native hold succeeded or the simulation budget was exhausted."""


class MapRecorder:
    """Store the arrays the upstream interface sends to its visualizer."""

    def __init__(self, output: Path):
        self.output = output
        self.count = 0

    def visualize(self, info: dict) -> None:
        self.count += 1
        arrays = {k: v for k, v in info.items() if isinstance(v, np.ndarray)}
        arrays["trajectory_xyz"] = np.asarray([p[0] for p in info["traj_world"]])
        np.savez_compressed(self.output / f"maps-{self.count:03d}.npz", **arrays)


class RoboTLEnv:
    """Translate VoxPoser's Cartesian pose and binary gripper API to native servos.

    Object geometry is privileged collision-mesh geometry. A fixed scalar closure
    bends each digit's proximal, middle and distal joints, preserving lateral
    joints as RoboTL's LEAP teleoperation does. No task-specific grasp IK is used.
    """

    def __init__(self, fixture: Path, task: str, output: Path, seed: int):
        self.model = load_scene(fixture)
        self.data = fixture_start(self.model, fixture)
        self.ids = resolve_ids(self.model, RunInfo.model_validate_json((fixture / "scene.json").read_text()))
        factory, stage, self.duration, self.instruction = TASKS[task]
        self.task = task
        self.objective = factory(self.model, self.data, self.ids, stage)
        self.episode = NativeEpisode(self.model, self.data, self.ids,
                                    RolloutBudget(batch=1, threads=1, horizon_s=1,
                                                  replan_s=.25, hold_s=5), seed=seed)
        self.episode.objective = self.objective
        self.status = "program_finished_without_hold"
        self.workspace_bounds_min = np.array([-.15, -.6, 0.0])
        self.workspace_bounds_max = np.array([.9, .6, 1.0])
        self.visualizer = MapRecorder(output)
        self.open_control = self.data.ctrl.copy()
        self.closed_control = self.open_control.copy()
        for digit in self.ids.digits:
            for actuator in (digit.servos.proximal, digit.servos.middle, digit.servos.distal):
                self.closed_control[actuator] = np.clip(
                    1.2 * self.episode.gear[actuator], *self.model.actuator_ctrlrange[actuator])
        self.scratch = mujoco.MjData(self.model)
        mujoco.mj_copyData(self.scratch, self.model, self.data)
        self.scratch.qpos[self.episode.qadr[7:]] = self.closed_control[7:] / self.episode.gear[7:]
        mujoco.mj_forward(self.model, self.scratch)
        palm_rotation = self.scratch.site_xmat[self.ids.palm].reshape(3, 3)
        self.tcp_offset = palm_rotation.T @ (
            pad_points(self.ids.digits, self.scratch).mean(axis=0) - self.scratch.site_xpos[self.ids.palm])
        self.gripper = 1.0
        self.initial_pose = np.r_[self.get_ee_pos(), self.get_ee_quat()]
        self.actions = []
        self.episode._record(self.objective)

    def get_ee_pos(self) -> np.ndarray:
        rotation = self.data.site_xmat[self.ids.palm].reshape(3, 3)
        return self.data.site_xpos[self.ids.palm] + rotation @ self.tcp_offset

    def get_ee_quat(self) -> np.ndarray:
        return Rotation.from_matrix(self.data.site_xmat[self.ids.palm].reshape(3, 3)).as_quat(scalar_first=True)

    def get_last_gripper_action(self) -> float:
        return self.gripper

    def get_object_names(self) -> list[str]:
        if self.task in ("hammer", "screwdriver"):
            end = "head" if self.task == "hammer" else "shaft"
            return [self.task, f"{self.task} handle", f"{self.task} {end}"]
        return ["mustard bottle" if self.task == "mustard" else "apple"]

    def get_3d_obs_by_name(self, name: str) -> tuple[np.ndarray, np.ndarray]:
        normalized = name.lower().replace("_", " ")
        if normalized not in self.get_object_names():
            raise ValueError(f"Unknown object {name!r}; available objects: {self.get_object_names()}")
        geoms = self.ids.objects
        if normalized.endswith(" handle"):
            geoms = [g for g in geoms if self.model.geom_bodyid[g] == self.ids.obj]
        elif normalized.endswith((" head", " shaft")):
            geoms = [g for g in geoms if self.model.geom_bodyid[g] != self.ids.obj]
        points = mesh_points(self.model, self.data, geoms)
        normals = points - points.mean(axis=0)
        normals /= np.maximum(np.linalg.norm(normals, axis=1, keepdims=True), 1e-9)
        return points, normals

    def get_scene_3d_obs(self, ignore_robot=False, ignore_grasped_obj=False):
        points = mesh_points(self.model, self.data, self.ids.objects)
        x, y = np.meshgrid(np.linspace(-.15, .9, 60), np.linspace(-.6, .6, 60))
        floor = np.c_[x.ravel(), y.ravel(), np.zeros(x.size)]
        points = np.concatenate([points, floor])
        return points, np.tile([0., 0., 1.], (len(points), 1))

    def advance(self, control: np.ndarray) -> None:
        self.data.ctrl[:] = control
        mujoco.mj_step(self.model, self.data)
        self.episode._record(self.objective)
        if self.episode.lifted_clock.held_for(self.data.time) >= 5:
            self.status = "success"
            raise TrialEnded
        if self.data.time - self.episode.start_time >= self.duration:
            self.status = "timeout"
            raise TrialEnded

    def apply_action(self, action: np.ndarray):
        action = np.asarray(action, dtype=float)
        rotation = Rotation.from_quat(action[3:7], scalar_first=True).as_matrix()
        position = action[:3] - rotation @ self.tcp_offset
        model, data, scratch = self.model, self.data, self.scratch
        mujoco.mj_copyData(scratch, model, data)
        qadr = self.episode.qadr[:7]
        lower, upper = model.jnt_range[self.episode.joints[:7]].T

        def residual(q):
            scratch.qpos[qadr] = q
            mujoco.mj_kinematics(model, scratch)
            turn = Rotation.from_matrix(rotation @ scratch.site_xmat[self.ids.palm].reshape(3, 3).T).as_rotvec()
            return np.r_[scratch.site_xpos[self.ids.palm] - position, .08 * turn]

        solved = least_squares(residual, np.clip(data.qpos[qadr], lower + 1e-8, upper - 1e-8),
                               bounds=(lower, upper), max_nfev=100)
        self.gripper = float(np.clip(action[7], 0, 1))
        target = self.gripper * self.open_control + (1 - self.gripper) * self.closed_control
        target[:7] = solved.x * self.episode.gear[:7]
        start = data.ctrl.copy()
        duration = max(.3, float(np.max(np.abs(target[:7] - start[:7]) / (ARM_SPEED * .25))) * 1.875,
                       float(np.max(np.abs(target[7:] - start[7:]))) / .6)
        steps = max(1, round(duration / model.opt.timestep))
        self.actions.append(dict(time_s=float(data.time), target=action.tolist(),
                                 ik_residual=float(np.linalg.norm(solved.fun)), duration_s=duration))
        for fraction in np.arange(1, steps + 1) / steps:
            blend = fraction**3 * (10 - 15 * fraction + 6 * fraction**2)
            self.advance(start + blend * (target - start))
        return None, 0.0, False

    def move_to_pose(self, pose, speed=None):
        return self.apply_action(np.r_[pose, self.gripper])

    def reset_to_default_pose(self):
        return self.move_to_pose(self.initial_pose)

    def open_gripper(self):
        return self.apply_action(np.r_[self.get_ee_pos(), self.get_ee_quat(), 1.0])

    def close_gripper(self):
        return self.apply_action(np.r_[self.get_ee_pos(), self.get_ee_quat(), 0.0])


def main() -> None:
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--task", choices=TASKS, required=True)
    parser.add_argument("--fixture", type=Path, required=True)
    parser.add_argument("--output", type=Path, required=True)
    parser.add_argument("--seed", type=int, default=0)
    args = parser.parse_args()
    args.output.mkdir(parents=True, exist_ok=False)
    np.random.seed(args.seed)
    openai.api_key = os.environ["OPENAI_API_KEY"]
    config = get_config("rlbench")
    config["lmp_config"]["env"]["visualize"] = True
    for cfg in config["lmp_config"]["lmps"].values():
        cfg["model"] = "gpt-4.1"
    env = RoboTLEnv(args.fixture, args.task, args.output, args.seed)
    (args.output / "config.json").write_text(json.dumps({
        "voxposer": config, "task": args.task, "instruction": env.instruction,
        "source_revision": "ff60eaad16a044591e2a3119f6f499449869b8e7", "seed": args.seed,
        "scene_sha256": hashlib.sha256((args.fixture / "scene.mjb").read_bytes()).hexdigest(),
        "adapter_sha256": hashlib.sha256(Path(__file__).read_bytes()).hexdigest(),
        "mujoco": mujoco.__version__, "numpy": np.__version__,
        "workspace_min": env.workspace_bounds_min.tolist(), "workspace_max": env.workspace_bounds_max.tolist(),
        "tcp_offset_palm": env.tcp_offset.tolist(), "open_control": env.open_control.tolist(),
        "closed_control": env.closed_control.tolist(),
        "adaptations": ["Native MuJoCo Panda+LEAP fixture, not RLBench Panda gripper",
                        "Privileged collision vertices with radial approximate normals",
                        "Bounded arm IK and fixed scalar hand closure; no grasp proposals or Guided MPC",
                        "Native final-pose acceptance measured from episode start; five seconds hold after program"],
    }, indent=2))
    generated = []
    original = LMP._cached_api_call

    def record_reply(lmp, **kwargs):
        prompt = kwargs["prompt"]
        code = original(lmp, **kwargs)
        generated.append(dict(generator=lmp._name, prompt=prompt, code=code))
        (args.output / "generated-code.json").write_text(json.dumps(generated, indent=2))
        return code

    began = time.monotonic()
    try:
        with patch.object(LMP, "_cached_api_call", record_reply):
            lmps, _ = setup_LMP(env, config, debug=False)
            set_lmp_objects(lmps, env.get_object_names())
            lmps["plan_ui"](env.instruction)
            for _ in range(round(5.1 / env.model.opt.timestep)):
                env.advance(env.data.ctrl.copy())
    except TrialEnded:
        pass
    except Exception as error:
        env.status = "program_error"
        (args.output / "error.json").write_text(json.dumps({"type": type(error).__name__, "message": str(error)}))
        raise
    finally:
        env.episode.wall_s = time.monotonic() - began
        (args.output / "actions.json").write_text(json.dumps(env.actions, indent=2))
        env.episode.save(args.output, env.status)


if __name__ == "__main__":
    main()
