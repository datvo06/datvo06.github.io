# Block 1: planner
composer("grasp the mustard bottle")
composer("move to 6cm above the table while keeping the mustard bottle upright")

# Block 2: composer
movable = parse_query_obj('gripper')
affordance_map = get_affordance_map('a point at the center of the mustard bottle')
gripper_map = get_gripper_map('open everywhere except 1cm around the mustard bottle')
execute(movable, affordance_map=affordance_map, gripper_map=gripper_map)

# Block 3: parse_query_obj
gripper = detect('gripper')
ret_val = gripper

# Block 4: get_affordance_map
affordance_map = get_empty_affordance_map()
mustard_bottle = parse_query_obj('mustard bottle')
x, y, z = mustard_bottle.position
affordance_map[x, y, z] = 1
ret_val = affordance_map

# Block 5: get_gripper_map
gripper_map = get_empty_gripper_map()
# open everywhere
gripper_map[:, :, :] = 1
# close when 1cm around the mustard bottle
mustard_bottle = parse_query_obj('mustard bottle')
set_voxel_by_radius(gripper_map, mustard_bottle.position, radius_cm=1, value=0)
ret_val = gripper_map

# Block 6: parse_query_obj
mustard_bottle = detect('mustard bottle')
ret_val = mustard_bottle

# Block 7: parse_query_obj
mustard_bottle = detect('mustard bottle')
ret_val = mustard_bottle

# Block 8: composer
movable = parse_query_obj('gripper')
affordance_map = get_affordance_map('a point 6cm above the table')
rotation_map = get_rotation_map('keep the mustard bottle upright')
execute(movable, affordance_map=affordance_map, rotation_map=rotation_map)

# Block 9: parse_query_obj
gripper = detect('gripper')
ret_val = gripper

# Block 10: get_affordance_map
affordance_map = get_empty_affordance_map()
table = parse_query_obj('table')
(min_x, min_y, min_z), (max_x, max_y, max_z) = table.aabb
center_x, center_y, center_z = table.position
# 6cm above so we add to z-axis
x = center_x
y = center_y
z = max_z + cm2index(6, 'z')
affordance_map[x, y, z] = 1
ret_val = affordance_map

# Block 11: get_rotation_map
rotation_map = get_empty_rotation_map()
mustard_bottle = parse_query_obj('mustard bottle')
# Upright means gripper z-axis aligns with world z-axis (up)
upright_quat = vec2quat([0, 0, 1])
rotation_map[:, :, :] = upright_quat
ret_val = rotation_map

# Block 12: parse_query_obj
table = detect('table')
ret_val = table

# Block 13: parse_query_obj
mustard_bottle = detect('mustard bottle')
ret_val = mustard_bottle
