# Block 11: get_rotation_map
rotation_map = get_empty_rotation_map()
hammer = parse_query_obj('hammer')
handle_dir = hammer.handle_direction  # assumed to be a normalized vector
upright_quat = vec2quat([0, 0, 1])  # z-up
# rotate so z-axis aligns with handle direction, then rotate 180 deg around x to make head above handle
align_handle_quat = vec2quat(handle_dir)
rotation_delta = euler2quat(np.pi, 0, 0)
target_rotation = qmult(align_handle_quat, rotation_delta)
rotation_map[:, :, :] = target_rotation
ret_val = rotation_map

# Block 14: get_affordance_map

(min_x, min_y, min_z), (max_x, max_y, max_z) = table.aabb
center_x, center_y, center_z = table.position
# 6cm above so we add to z-axis
x = center_x

z = max_z + cm2index(6, 'z')
affordance_map[x, y, z] = 1
ret_val = affordance_map
